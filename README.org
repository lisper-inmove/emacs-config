* 用Systemctl管理Emacs
#+begin_src shell
  [Unit]
  Description=Emacs Daemon (GUI-capable, conda env)
  Documentation=man:emacs(1) man:emacsclient(1)
  After=graphical-session.target
  PartOf=graphical-session.target

  [Service]
  Environment=ENABLE_GUI=true
  Type=simple
  # 导入图形会话变量，保证能建 GUI 窗口
  ExecStartPre=/usr/bin/systemctl --user import-environment DISPLAY WAYLAND_DISPLAY XAUTHORITY XDG_RUNTIME_DIR DBUS_SESSION_BUS_ADDRESS
  Environment=CHATGPT_SHELL_DS_Emacs01=Deep-Seek-Key

  # >>> 在 myenv 环境里启动 emacs daemon（两种写法二选一） <<<
  # 1) 用环境名：
  # ExecStart=/home/inmove/.miniconda3/bin/conda run --no-capture-output -n python311 /home/inmove/.bin/emacs --fg-daemon
  # 2) 用环境路径（更可靠）：
  ExecStart=/home/inmove/.miniconda3/bin/conda run --no-capture-output -p /home/inmove/.miniconda3/envs/python311 /home/inmove/.bin/emacs --fg-daemon

  # 停止时也可用系统里的 emacsclient 即可（只是发指令，不影响 daemon 的环境）
  ExecStop=/home/inmove/.bin/emacsclient --eval "(kill-emacs)"
  Restart=on-failure

  [Install]
  WantedBy=graphical-session.target

#+end_src
